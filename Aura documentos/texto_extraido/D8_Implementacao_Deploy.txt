import { supabase } from '../lib/supabase'

export function useAuth() {
  const queryClient = useQueryClient()
  const { user, isAuthenticated, isLoading } = useAuthStore()
  const signInMutation = useMutation({
    mutationFn: async ({ email, password }: { email: string; password: string }) => {
      const { data, error } = await supabase.auth.signInWithPassword({ email, password }
      if (error) throw error
      return data
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['profile'] })
    },
  })
  const signUpMutation = useMutation({
    mutationFn: async ({ email, password, userData }: any) => {
      const { data, error } = await supabase.auth.signUp({
        email,
        password,
        options: { data: userData },
      })
      if (error) throw error
      return data
    },
  })

import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { supabase } from '../lib/supabase'
const EVENTS_KEY = 'events'
export function useEvents(filters?: Record<string, any>) {
  return useQuery({
    queryKey: [EVENTS_KEY, filters],
    queryFn: async () => {
      let query = supabase
        .from('events')
        .select('*', { count: 'exact' })
        .eq('status', 'published')
        .order('start_date', { ascending: true })
      if (filters?.city) query = query.eq('venue_city', filters.city)
      if (filters?.category) query = query.eq('category', filters.category)
      if (filters?.search) query = query.ilike('title', `%${filters.search}%`)
      const { data, error, count } = await query
      if (error) throw error
      return { events: data || [], count: count || 0 }
    },
  })
}

import { useMutation, useQueryClient } from '@tanstack/react-query'
import { supabase } from '../lib/supabase'
export function useCheckin() {
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: async ({ ticketCode, eventId }: { ticketCode: string; eventId: string }
      const { data: ticket, error: ticketError } = await supabase
        .from('ticket_order_items')
        .select(`*, order:ticket_orders(event_id)`)
        .eq('ticket_code', ticketCode)
        .single()
      if (ticketError) throw new Error('Ingresso nao encontrado')
      if (ticket.order.event_id !== eventId) {
        throw new Error('Ingresso nao pertence a este evento')
      }
      if (ticket.checkin_at) {
        throw new Error('Ingresso ja utilizado em ' + new Date(ticket.checkin_at).toLoca
      }
      const { data, error } = await supabase
        .from('ticket_order_items')
        .update({
          checkin_at: new Date().toISOString(),
          checkin_by: (await supabase.auth.getUser()).data.user?.id,
          checkin_device: navigator.userAgent,
        })
        .eq('id', ticket.id)
        .select()
        .single()
      if (error) throw error
      return data
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['checkin-stats'] })
    },
  })
}
